function updateOffsetOptions() {
const segmentRegister =
document.getElementById("segmentRegister").value;
const offsetRegister =
document.getElementById("offsetRegister");
const options = {
CS: [
["IP", "IP - Instruction Pointer"]
],
DS: [
["BX", "BX - Base Register"],
["SI", "SI - Source Index"],
["DI", "DI - Destination Index"],
["Direct", "Direct Offset"]
],
SS: [
["SP", "SP - Stack Pointer"],
["BP", "BP - Base Pointer"]
],
ES: [
["DI", "DI - Destination Index (String Operations)"],
["BX", "BX - Base Register"],
["SI", "SI - Source Index"],
["Direct", "Direct Offset"]
]
};
offsetRegister.innerHTML = "";
options[segmentRegister].forEach(function(item) {
const option = document.createElement("option");
option.value = item[0];
option.textContent = item[1];
offsetRegister.appendChild(option);
});
checkCombination();
}
function checkCombination() {
const segmentRegister =
document.getElementById("segmentRegister").value;
const offsetRegister =
document.getElementById("offsetRegister").value;
const warning = document.getElementById("warning");
const common = {
CS: ["IP"],
DS: ["BX", "SI", "DI", "Direct"],
SS: ["SP", "BP"],
ES: ["DI"]
};
if (!common[segmentRegister].includes(offsetRegister)) {
warning.textContent =
"⚠️ This is not a common register combination. " +
"Please check the selected registers.";
warning.style.display = "block";
} else {
warning.style.display = "none";
}
}
function calculateAddress() {
const segment = document.getElementById("segment")
.value.trim();
const offset = document.getElementById("offset")
.value.trim();
const segReg = document.getElementById("segmentRegister").value;
const offReg = document.getElementById("offsetRegister").value;
const error = document.getElementById("error");
const result = document.getElementById("result");
error.style.display = "none";
const hexPattern = /^[0-9A-Fa-f]{1,4}$/;
if (!hexPattern.test(segment) || !hexPattern.test(offset)) {
error.textContent =
"Please enter valid hexadecimal values from 0000 to FFFF.";
error.style.display = "block";
result.style.display = "none";
return;
}
const segValue = parseInt(segment, 16);
const offValue = parseInt(offset, 16);
const shiftedSegment = segValue * 16;
const physicalAddress = shiftedSegment + offValue;
const hex = value => value.toString(16)
.toUpperCase()
.padStart(5, "0");
const segHex = segValue.toString(16).toUpperCase()
.padStart(4, "0");
const offHex = offValue.toString(16).toUpperCase()
.padStart(4, "0");
const shiftedHex = hex(shiftedSegment);
const physicalHex = hex(physicalAddress);
document.getElementById("answer").textContent =
physicalHex + "H";
document.getElementById("formula").innerHTML =
"<strong>Formula applied</strong><br>" +
segReg + " × 10H + " + offReg + "<br>" +
shiftedHex + "H + " + offHex + "H = " +
physicalHex + "H";
const steps = [
"<strong>Segment:</strong> " + segReg + " = " + segHex + "H",
"<strong>Offset:</strong> " + offReg + " = " + offHex + "H",
"<strong>Multiply segment by 10H:</strong> " +
segHex + "H × 10H = " + shiftedHex + "H",
"<strong>Add the offset:</strong> " +
shiftedHex + "H + " + offHex + "H",
"<strong>Physical address:</strong> " +
physicalHex + "H"
];
document.getElementById("steps").innerHTML =
steps.map((step, index) =>
'<div class="step">' +
'<span class="number">' + (index + 1) + '</span>' +
'<div>' + step + '</div></div>'
).join("");
result.style.display = "block";
result.scrollIntoView({ behavior: "smooth", block: "nearest" });
}
function clearCalculator() {
document.getElementById("segment").value = "";
document.getElementById("offset").value = "";
document.getElementById("result").style.display = "none";
document.getElementById("error").style.display = "none";
document.getElementById("warning").style.display = "none";
document.getElementById("segmentRegister").selectedIndex = 0;
updateOffsetOptions();
}
document.getElementById("segmentRegister")
.addEventListener("change", updateOffsetOptions);
document.getElementById("offsetRegister")
.addEventListener("change", checkCombination);
updateOffsetOptions();
